calibrate touch screen

	apt install xinput_calibratorn 
	cd /etc/X11/xorg.conf.d
	sudo nano 40-libinput.conf
	search for "touchscreen"
		check below and find Option "CalibrationMatrix" "1 0 0 0 1 0 0 0 1"
		change 3rd and 6th of "CalibrationMatrix" in decimal percentage to adjust x and y offset. 
		for screen now is "1.12 0 -0.22 0 1.12 -0.20 0 0 1"
